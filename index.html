# (c) script.js
cat > script.js <<'EOF'
(async () => {
  const { createFFmpeg, fetchFile } = FFmpeg;
  const ffmpeg = createFFmpeg({ log: true });

  const logEl = document.getElementById('log');
  const log = (msg) => {
    logEl.textContent += msg + '\n';
    logEl.scrollTop = logEl.scrollHeight;
  };

  const fileInput = document.getElementById('videoFile');
  const preview = document.getElementById('preview');
  const previewSection = document.getElementById('previewSection');
  const controlsSection = document.getElementById('controlsSection');
  const logSection = document.getElementById('logSection');
  const startInput = document.getElementById('startSec');
  const endInput = document.getElementById('endSec');
  const trimBtn = document.getElementById('trimBtn');

  fileInput.addEventListener('change', async () => {
    const file = fileInput.files[0];
    if (!file) return;
    previewSection.style.display = 'none';
    controlsSection.style.display = 'none';
    logSection.style.display = 'none';
    logEl.textContent = '';

    const url = URL.createObjectURL(file);
    preview.src = url;
    previewSection.style.display = 'block';
    controlsSection.style.display = 'block';

    preview.onloadedmetadata = () => {
      endInput.value = Math.floor(preview.duration);
    };
  });

  trimBtn.addEventListener('click', async () => {
    const file = fileInput.files[0];
    if (!file) return alert('Select a video first');

    const start = parseFloat(startInput.value);
    const end = parseFloat(endInput.value);
    if (isNaN(start) || isNaN(end) || start >= end) {
      return alert('Enter a valid start/end time');
    }

    trimBtn.disabled = true;
    logSection.style.display = 'block';
    log('Loading FFmpeg core…');
    if (!ffmpeg.isLoaded) await ffmpeg.load();

    const nameIn = 'input' + file.name.replace(/\W/g,'');
    ffmpeg.FS('writeFile', nameIn, await fetchFile(file));

    const nameOut = 'clip_' + Date.now() + '.mp4';
    const args = ['-y','-i',nameIn,'-ss',`${start}`,'-to',`${end}`,'-c','copy',nameOut];
    log(`Running: ffmpeg ${args.join(' ')}`);
    const { rc } = await ffmpeg.run(...args);
    if (rc !== 0) {
      log('❌ FFmpeg failed');
      trimBtn.disabled = false;
      return;
    }

    const data = ffmpeg.FS('readFile', nameOut);
    const blob = new Blob([data.buffer], {type:'video/mp4'});
    const dlUrl = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = dlUrl;
    a.download = nameOut;
    a.click();

    log('✅ Done! File downloaded.');
    trimBtn.disabled = false;
  });
})();
EOF
